const openai = new OpenAI({

    try {
        const response = await openai.chat.completions.create({
            model: "gpt-4-turbo-preview",
            messages: [
                {
                    role: "user",
                    content: prompt
                }
            ],
            max_tokens: maxTokens,
            temperature: 0.7,
        });
        
        return response.choices[0].message.content;
    } catch (error) {
        console.error('GPT-4 API Error:', error);
        throw error;
    }
}

// Function Calling Example
async function callGPTWithFunctions(prompt) {
    const response = await openai.chat.completions.create({
        model: "gpt-4-turbo-preview",
        messages: [{ role: "user", content: prompt }],
        functions: [
            {
                name: "analyze_data",
                description: "Analyze business data",
                parameters: {
                    type: "object",
                    properties: {
                        metric: { type: "string" },
                        value: { type: "number" }
                    }
                }
            }
        ]
    });
    
    return response;
}</code></pre>

    constructor() {
        this.providers = {
            claude: new ClaudeProvider(),
            gpt4: new GPT4Provider(),
            gemini: new GeminiProvider(),
            deepseek: new DeepSeekProvider(),
            llama: new LlamaProvider(),
            mistral: new MistralProvider()
        };
    }

    async route(prompt, options = {}) {
        const {
            model = 'claude',
            fallbacks = ['gpt4', 'gemini'],
            maxTokens = 1000,
            temperature = 0.7,
            timeout = 30000
        } = options;

        // Try primary model
        try {
            return await this.callWithTimeout(
                this.providers[model].generate(prompt, { maxTokens, temperature }),
                timeout
            );
        } catch (error) {
            console.warn(`Primary model ${model} failed:`, error.message);
            
            // Try fallbacks
            for (const fallback of fallbacks) {
                try {
                    console.log(`Trying fallback: ${fallback}`);
                    return await this.callWithTimeout(
                        this.providers[fallback].generate(prompt, { maxTokens, temperature }),
                        timeout
                    );
                } catch (fallbackError) {
                    console.warn(`Fallback ${fallback} failed:`, fallbackError.message);
                }
            }
            
            throw new Error('All AI providers failed');
        }
    }

    async callWithTimeout(promise, timeout) {
        return Promise.race([
            promise,
            new Promise((_, reject) => 
                setTimeout(() => reject(new Error('Request timeout')), timeout)
            )
        ]);
    }

    // Load balancing
    async distributedCall(prompts, options = {}) {
        const models = Object.keys(this.providers);
        const results = [];

        for (let i = 0; i < prompts.length; i++) {
            const model = models[i % models.length];
            try {
                const result = await this.route(prompts[i], { 
                    ...options, 
                    model,
                    fallbacks: models.filter(m => m !== model)
                });
                results.push({ success: true, result, model });
            } catch (error) {
                results.push({ success: false, error: error.message, prompt: prompts[i] });
            }
        }

        return results;
    }
}
